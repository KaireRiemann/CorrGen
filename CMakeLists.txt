cmake_minimum_required(VERSION 3.16)
project(corrgen_cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_path(GLPK_INCLUDE_DIR glpk.h REQUIRED)
find_library(GLPK_LIBRARY glpk REQUIRED)
add_library(corrgen src/corridor.cpp src/corridor_check.cpp src/generator.cpp src/solver.cpp src/sdlp_adapter.cpp src/path.cpp src/voxel_planner.cpp)
target_include_directories(corrgen PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${GLPK_INCLUDE_DIR})
target_link_libraries(corrgen PUBLIC Eigen3::Eigen PRIVATE ${GLPK_LIBRARY})
target_compile_options(corrgen PRIVATE -Wall -Wextra -Wpedantic)
find_path(HIGHS_INCLUDE_DIR Highs.h PATH_SUFFIXES highs)
find_library(HIGHS_LIBRARY highs)
if(HIGHS_INCLUDE_DIR AND HIGHS_LIBRARY)
  target_compile_definitions(corrgen PRIVATE CORRGEN_HIGHS=1)
  target_include_directories(corrgen SYSTEM PRIVATE ${HIGHS_INCLUDE_DIR})
  target_link_libraries(corrgen PRIVATE ${HIGHS_LIBRARY})
endif()
find_path(PROXQP_INCLUDE_DIR proxsuite/proxqp/dense/dense.hpp)
if(PROXQP_INCLUDE_DIR)
  target_sources(corrgen PRIVATE src/proxqp_adapter.cpp)
  target_include_directories(corrgen SYSTEM PRIVATE ${PROXQP_INCLUDE_DIR})
  target_compile_definitions(corrgen PRIVATE CORRGEN_PROXQP=1)
endif()
add_executable(corrgen_benchmark apps/benchmark.cpp)
target_link_libraries(corrgen_benchmark PRIVATE corrgen)
add_executable(corrgen_replay apps/replay.cpp)
target_link_libraries(corrgen_replay PRIVATE corrgen)
include(CTest)
if(BUILD_TESTING)
  add_executable(corrgen_tests tests/test_corrgen.cpp)
  target_link_libraries(corrgen_tests PRIVATE corrgen)
  add_test(NAME corrgen_tests COMMAND corrgen_tests)
endif()
install(TARGETS corrgen corrgen_benchmark ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)

option(CORRGEN_BUILD_ROS1 "Build ROS1 A* demo adapter" OFF)
if(CORRGEN_BUILD_ROS1 OR CATKIN_DEVEL_PREFIX)
  find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs visualization_msgs std_msgs geometry_msgs)
  catkin_package(INCLUDE_DIRS include LIBRARIES corrgen CATKIN_DEPENDS roscpp sensor_msgs nav_msgs visualization_msgs std_msgs geometry_msgs DEPENDS Eigen3)
  find_package(Threads REQUIRED)
  find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
  add_executable(corrgen_worker apps/worker.cpp)
  target_include_directories(corrgen_worker PRIVATE ${JSON_INCLUDE_DIR})
  target_link_libraries(corrgen_worker PRIVATE corrgen Threads::Threads)
  add_executable(corrgen_demo_node ros/demo_node.cpp)
  target_include_directories(corrgen_demo_node PRIVATE ${catkin_INCLUDE_DIRS} ${JSON_INCLUDE_DIR})
  target_link_libraries(corrgen_demo_node PRIVATE corrgen ${catkin_LIBRARIES})
  install(TARGETS corrgen_demo_node corrgen_worker RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  install(DIRECTORY launch rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
endif()

set(CORRGEN_REAL_PLANNER_DIR "" CACHE PATH "Optional real_planner general_planner directory for SDLP comparison")
if(CORRGEN_REAL_PLANNER_DIR)
  if(BUILD_TESTING)
    target_compile_definitions(corrgen_tests PRIVATE CORRGEN_TEST_SDLP=1)
  endif()
  target_compile_definitions(corrgen PRIVATE CORRGEN_SDLP_SOURCE="${CORRGEN_REAL_PLANNER_DIR}/src/utils/sdlp.cpp")
  target_include_directories(corrgen PRIVATE ${CORRGEN_REAL_PLANNER_DIR}/include)
  add_executable(corrgen_compare_sdlp apps/compare_sdlp.cpp)
  target_compile_definitions(corrgen_compare_sdlp PRIVATE CORRGEN_SDLP_SOURCE="${CORRGEN_REAL_PLANNER_DIR}/src/utils/sdlp.cpp")
  target_include_directories(corrgen_compare_sdlp PRIVATE ${CORRGEN_REAL_PLANNER_DIR}/include ${GLPK_INCLUDE_DIR})
  target_link_libraries(corrgen_compare_sdlp PRIVATE corrgen ${GLPK_LIBRARY})
endif()
